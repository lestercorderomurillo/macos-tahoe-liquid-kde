cmake_minimum_required(VERSION 3.22)
project(GlassMaterialProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(QT_MAJOR_VERSION 6)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH} ${ECM_KDE_MODULE_DIR})
include(KDEInstallDirs)
include(KDECMakeSettings)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus OpenGL Svg)
find_package(KF6 REQUIRED COMPONENTS Config ConfigWidgets CoreAddons I18n KCMUtils WidgetsAddons)
find_package(KWinDBusInterface QUIET)
if(NOT KWinDBusInterface_FOUND)
    message(STATUS "GLASS_KCM_SDK_MISSING")
    return()
endif()
add_subdirectory("${GLASS_SOURCE}/src/kcm" kcm)
add_executable(glass-kcm-probe kcm.cpp)
target_link_libraries(glass-kcm-probe PRIVATE Qt6::Widgets KF6::KCMUtils KF6::CoreAddons KF6::ConfigGui KF6::WidgetsAddons)

add_executable(glass-shader-probe shader.cpp)
target_link_libraries(glass-shader-probe PRIVATE Qt6::Gui Qt6::OpenGL Qt6::Svg)

target_include_directories(glass-shader-probe PRIVATE "${GLASS_SOURCE}/src")
