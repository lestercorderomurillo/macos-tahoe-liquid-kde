cmake_minimum_required(VERSION 3.16)
project(MacTahoeMenuStyle LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6WindowSystem REQUIRED)
add_library(mactahoestyle MODULE menu_style.cpp)
target_link_libraries(mactahoestyle PRIVATE Qt6::Widgets KF6::WindowSystem)
set(MENU_STYLE_PLUGIN_DIR "lib/qt6/plugins/styles" CACHE PATH "Qt style plugin destination")
install(TARGETS mactahoestyle LIBRARY DESTINATION ${MENU_STYLE_PLUGIN_DIR})
